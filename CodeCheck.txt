var target = "Jul 4, 2026";

// Find which column index is "Created" and which is "Modified"
var headers = document.querySelectorAll("thead th");
var createdIdx = -1, modifiedIdx = -1;
headers.forEach(function(th, idx) {
    var text = th.innerText.trim();
    if (text.includes("Created")) createdIdx = idx;
    if (text.includes("Modified")) modifiedIdx = idx;
});
console.log("Created column index:", createdIdx, "| Modified column index:", modifiedIdx);

var rows = document.querySelectorAll("tbody tr");
var createdCount = 0, modifiedCount = 0, bothCount = 0, eitherCount = 0;

rows.forEach(function(row) {
    var cells = row.querySelectorAll("td");
    if (cells.length === 0) return;

    var inCreated  = createdIdx  >= 0 && cells[createdIdx]  && cells[createdIdx].innerText.includes(target);
    var inModified = modifiedIdx >= 0 && cells[modifiedIdx] && cells[modifiedIdx].innerText.includes(target);

    if (inCreated) createdCount++;
    if (inModified) modifiedCount++;
    if (inCreated && inModified) bothCount++;
    if (inCreated || inModified) eitherCount++;
});

console.log("Created  = " + target + " :", createdCount);
console.log("Modified = " + target + " :", modifiedCount);
console.log("Both (created AND modified today):", bothCount);
console.log("Either (unique rows, no double count):", eitherCount);
